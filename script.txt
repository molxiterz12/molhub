local function SafeCall(fn, label)
    return function(...)
        local ok, err = pcall(fn, ...)

-- PROTECT GUI (FIXED - fallback ke CoreGui + PlayerGui)
-- ============================================================
local function protectGui(sg)
    -- Coba protect dulu
    pcall(function()
        if syn and syn.protect_gui then
            syn.protect_gui(sg)
        end
    end)
    -- Coba gethui
    local parented = false
    if gethui then
        pcall(function()
            local hui = gethui()
            if hui then
                sg.Parent = hui
                parented = true
            end
        end)
    end
    -- Fallback ke CoreGui
    if not parented then
        pcall(function()
            sg.Parent = game:GetService("CoreGui")
            parented = true
        end)
    end
    -- Fallback terakhir ke PlayerGui
    if not parented or not sg.Parent then
        pcall(function()
            local pg = LocalPlayer:FindFirstChildOfClass("PlayerGui")
            if not pg then
                pg = LocalPlayer:WaitForChild("PlayerGui", 5)
            end
            if pg then
                sg.Parent = pg
            end
        end)
    end
end

local function mk(class, props)
    local i = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then
            pcall(function() i[k] = v end)
        end
    end
    if props and props.Parent then i.Parent = props.Parent end
    return i
end
local function corner(i, r)
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, r or 8); c.Parent = i; return c
end
local function stroke(i, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color or Theme.Border; s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border; s.Parent = i; return s
end
local function tween(i, t, p)
    TweenService:Create(i, TweenInfo.new(t or 0.18, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), p):Play()
end

        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then

            main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)

            tween(tab.btn, 0.15, {BackgroundColor3 = tn == name and Theme.Element or Theme.Sidebar})
            tab.btnLbl.TextColor3 = tn == name and Theme.Text or Theme.SubText

        b.MouseLeave:Connect(function() if activeTab ~= name then tween(b, 0.12, {BackgroundColor3=Theme.Sidebar}) end end)
        b.MouseButton1Click:Connect(function() showTab(name) end)
        return container
    end

    return setmetatable({sg=sg, main=main, sidebar=sidebar, content=content, addTab=addTab, showTab=showTab, tabs=tabs}, Window)
end

        knob.Position = UDim2.new(rel, -6, 0.5, -6)

        dragging = true

local currentTarget = nil
local lastFire = 0

local function getHrp(p)
    if not p or not p.Character then return nil end
    return p.Character:FindFirstChild("HumanoidRootPart")
end
local function IsEnemy(player)
    if player == LocalPlayer then return false end
    if not Config.TeamCheck then return true end
    if player.Team and LocalPlayer.Team then return player.Team ~= LocalPlayer.Team end
    return true
end
local function GetAimPart(player)
    if not player or not player.Character then return nil end
    local char = player.Character
    local part = char:FindFirstChild(Config.AimPart)
    if part and part:IsA("BasePart") then return part end
    for _, name in ipairs({"Head","UpperTorso","Torso","HumanoidRootPart","LowerTorso"}) do
        local p = char:FindFirstChild(name)
        if p and p:IsA("BasePart") then return p end
    end
    return nil
end
local function HasLineOfSight(part)
    if not part then return false end
    local origin = Camera.CFrame.Position
    local dir = part.Position - origin
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local ignore = {LocalPlayer.Character, Camera}
    if part.Parent then table.insert(ignore, part.Parent) end
    params.FilterDescendantsInstances = ignore
    params.IgnoreWater = true
    local result = workspace:Raycast(origin, dir, params)
    return result == nil or result.Instance == part or result.Instance:IsDescendantOf(part.Parent)
end
local function IsTargetValid(player)
    if not player or not player.Parent then return false end
    if not IsEnemy(player) then return false end
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    local part = GetAimPart(player)
    if not part then return false end
    if (part.Position - Camera.CFrame.Position).Magnitude > 5000 then return false end
    if Config.VisibleCheck and not HasLineOfSight(part) then return false end
    return true
end
local function AcquireTarget()
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local best, bestDist = nil, Config.FOV
    for _, player in ipairs(Players:GetPlayers()) do
        if IsEnemy(player) and player.Character then
            local part = GetAimPart(player)
            if part then
                local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen and sp.Z > 0 then
                    local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d < bestDist then
                        if not Config.VisibleCheck or HasLineOfSight(part) then
                            bestDist = d; best = player
                        end
                    end
                end
            end
        end
    end
    return best
end
local function GetPredictedPosition(part)
    if not Config.UsePrediction or not part then return part.Position end
    local char = part.Parent
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return part.Position end
    local vel = hrp.AssemblyLinearVelocity or hrp.Velocity or Vector3.zero
    local dist = (part.Position - Camera.CFrame.Position).Magnitude
    local travelTime = dist / 1000
    return part.Position + vel * travelTime * Config.Prediction
end
local function AimAt(player)
    local part = GetAimPart(player)
    if not part then return end
    local targetPos = GetPredictedPosition(part)
    local goal = CFrame.new(Camera.CFrame.Position, targetPos)
    if Config.Smoothness >= 1 then
        Camera.CFrame = goal
    else
        Camera.CFrame = Camera.CFrame:Lerp(goal, Config.Smoothness)
    end
end

end

local KILLER_WORDS = {"killer","monster","zombie","slasher","hunter","stalker","chaser","entity","beast","demon","ghost","phantom","murderer","assassin","vampire","werewolf"}
local function nameMatchesKiller(name)
    if not name then return false end
    local n = string.lower(tostring(name))
    for _, kw in ipairs(KILLER_WORDS) do if string.find(n, kw, 1, true) then return true end end
    return false
end
local function isKillerPlayer(player)
    if not player then return false end
    if nameMatchesKiller(player.Name) then return true end
    if nameMatchesKiller(player.DisplayName) then return true end
    local ls = player:FindFirstChild("leaderstats")
    if ls then
        for _, v in ipairs(ls:GetChildren()) do
            if nameMatchesKiller(v.Name) then return true end
            if v:IsA("StringValue") and nameMatchesKiller(v.Value) then return true end
        end
    end
    return false
end

local function findRandomByKeywords(keywords)
    local found, seen = {}, {}
    local function matchName(n)
        n = string.lower(n or "")
        for _, kw in ipairs(keywords) do if string.find(n, kw, 1, true) then return true end end
        return false
    end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and matchName(obj.Name) and not seen[obj] then
            seen[obj] = true; table.insert(found, obj)
        end
        if obj:IsA("Model") and matchName(obj.Name) then
            local p = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
            if p and not seen[p] then seen[p] = true; table.insert(found, p) end
        end
    end
    if #found == 0 then return nil end
    return found[math.random(1, #found)]
end

local flyConn, flyBV, flyBG
local flySavedWS, flySavedJP, flySavedAR
local function startFly()
    local char = LocalPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end
    flySavedWS = hum.WalkSpeed; flySavedJP = hum.JumpPower; flySavedAR = hum.AutoRotate
    hum.PlatformStand = true; hum.AutoRotate = false; hum.WalkSpeed = 0; hum.JumpPower = 0
    for _, m in ipairs(hrp:GetChildren()) do if m.Name=="MolFlyBV" or m.Name=="MolFlyBG" then m:Destroy() end end
    flyBV = Instance.new("BodyVelocity")
    flyBV.Name="MolFlyBV"; flyBV.MaxForce=Vector3.new(math.huge,math.huge,math.huge)
    flyBV.P=1250; flyBV.Velocity=Vector3.zero; flyBV.Parent=hrp
    flyBG = Instance.new("BodyGyro")
    flyBG.Name="MolFlyBG"; flyBG.MaxTorque=Vector3.new(math.huge,math.huge,math.huge)
    flyBG.P=100000; flyBG.D=1000; flyBG.CFrame=hrp.CFrame; flyBG.Parent=hrp
    if flyConn then flyConn:Disconnect() end
    flyConn = RunService.RenderStepped:Connect(SafeCall(function()
        if not states.Fly then return end
        if not hrp.Parent or not flyBV.Parent or not flyBG.Parent then return end
        hum.PlatformStand=true; hum.WalkSpeed=0; hum.JumpPower=0; hum.AutoRotate=false
        local camCF = Camera.CFrame
        local look = camCF.LookVector
        local right = Vector3.new(camCF.RightVector.X, 0, camCF.RightVector.Z).Unit
        local move = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move += look end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move -= look end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move -= right end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move += right end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move -= Vector3.new(0,1,0) end
        if move.Magnitude > 0.01 then
            flyBV.Velocity = move.Unit * Config.FlySpeed
        else
            flyBV.Velocity = Vector3.zero
        end
        flyBG.CFrame = camCF
    end, "FlyLoop"))
end
local function stopFly()
    if flyConn then flyConn:Disconnect(); flyConn=nil end
    if flyBV then flyBV:Destroy(); flyBV=nil end
    if flyBG then flyBG:Destroy(); flyBG=nil end
    local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.PlatformStand=false
        hum.WalkSpeed = flySavedWS or 16
        hum.JumpPower = flySavedJP or 50
        hum.AutoRotate = flySavedAR ~= false
    end
end

                    local boxColor = (isK and Config.ESPKiller) and Config.ESPKillerCol or Config.ESPBoxColor
                    if Config.ESPBox then
                        esp.Box.Size = Vector2.new(width, height)
                        esp.Box.Position = Vector2.new(sp.X - width/2, headSp.Y)
                        esp.Box.Color = boxColor; esp.Box.Visible = true
                    else esp.Box.Visible = false end
                    if Config.ESPName then
                        esp.Name.Text = player.Name; esp.Name.Position = Vector2.new(sp.X, headSp.Y - 18)
                        esp.Name.Color = (isK and Config.ESPKiller) and Config.ESPKillerCol or Config.ESPNameColor
                        esp.Name.Visible = true
                    else esp.Name.Visible = false end
                    if Config.ESPHealth and hum then
                        esp.Health.Text = string.format("[%d HP]", math.floor(hum.Health))
                        esp.Health.Position = Vector2.new(sp.X, headSp.Y - 34)
                        esp.Health.Color = Config.ESPHealthColor; esp.Health.Visible = true
                    else esp.Health.Visible = false end
                    if Config.ESPDistance then
                        local dist = (hrp.Position - Camera.CFrame.Position).Magnitude
                        esp.Distance.Text = string.format("%dm", math.floor(dist))
                        esp.Distance.Position = Vector2.new(sp.X, headSp.Y + height + 4)
                        esp.Distance.Color = Config.ESPDistanceColor; esp.Distance.Visible = true
                    else esp.Distance.Visible = false end
                    if Config.ESPAntenna then
                        esp.Antenna.From = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y)
                        esp.Antenna.To = Vector2.new(sp.X, sp.Y)
                        esp.Antenna.Color = (isK and Config.ESPKiller) and Config.ESPKillerCol or Config.ESPAntennaCol
                        esp.Antenna.Visible = true
                    else esp.Antenna.Visible = false end
                else
                    esp.Box.Visible=false; esp.Name.Visible=false
                    esp.Health.Visible=false; esp.Distance.Visible=false; esp.Antenna.Visible=false
                end
            else
                esp.Box.Visible=false; esp.Name.Visible=false
                esp.Health.Visible=false; esp.Distance.Visible=false; esp.Antenna.Visible=false
            end
        end

        -- Skeleton
        if Config.ESPSkeleton then
            for player, _ in pairs(SkeletonLines) do
                if espEnemy(player) then
                    local isK = isKillerPlayer(player)
                    local col = (isK and Config.ESPKiller) and Config.ESPKillerCol or Config.ESPSkeletonCol
                    drawSkeleton(player, col)
                else
                    for _, l in ipairs(SkeletonLines[player]) do l.Visible = false end
                end
            end
        else
            for _, lines in pairs(SkeletonLines) do for _, l in ipairs(lines) do l.Visible = false end end
        end
    end, "ESP"))
end)

task.spawn(function()
    RunService.Heartbeat:Connect(SafeCall(function()
        -- Spinbot
        if states.Spinbot then
            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            if hrp then
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum then hum.AutoRotate = false end
                if not spinAV or not spinAV.Parent then startSpinbot()
                else spinAV.AngularVelocity = Vector3.new(0, Config.SpinbotSpeed, 0) end
            end
        else
            if spinAV then stopSpinbot() end
        end

        -- Speed
        local char = LocalPlayer.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                if Config.SpeedEnabled and not states.Fly then
                    if hum.WalkSpeed ~= Config.WalkSpeed then hum.WalkSpeed = Config.WalkSpeed end
                end
                if Config.Speed50 and not states.Fly then
                    if hum.WalkSpeed ~= Config.SpeedValue then hum.WalkSpeed = Config.SpeedValue end
                end
                if Config.JumpEnabled and not states.Fly then
                    if hum.JumpPower ~= Config.JumpPower then hum.JumpPower = Config.JumpPower end
                end
                if Config.HipHeight ~= 2 and hum.HipHeight ~= Config.HipHeight then
                    hum.HipHeight = Config.HipHeight
                end
            end
        end

        -- Lighting
        if Config.FullBright then
            Lighting.Brightness = 3
            Lighting.ClockTime = 12
            Lighting.Ambient = Color3.fromRGB(180,180,180)
            Lighting.OutdoorAmbient = Color3.fromRGB(180,180,180)
            Lighting.GlobalShadows = false
        end
        if Config.NoFog then
            Lighting.FogEnd = 100000
            Lighting.FogStart = 99999
        end
    end, "Main"))
end)
