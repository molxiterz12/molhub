local function tween(obj, time, props)
    local t = TweenService:Create(obj, TweenInfo.new(time or 0.2, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), props)
    t:Play(); return t
end
local function new(class, props, children)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do if k ~= "Parent" then inst[k] = v end end
    for _, ch in ipairs(children or {}) do ch.Parent = inst end
    if props and props.Parent then inst.Parent = props.Parent end
    return inst
end
local function corner(inst, r)
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, r or 10); c.Parent = inst; return c
end
local function stroke(inst, color, thickness)
    local s = Instance.new("UIStroke")
    s.Color = color or C.BORDER; s.Thickness = thickness or 2
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border; s.Parent = inst; return s
end
local function protectGui(sg)
    pcall(function()
        if syn and syn.protect_gui then syn.protect_gui(sg); sg.Parent = game:GetService("CoreGui")
        elseif gethui then sg.Parent = gethui()
        else sg.Parent = game:GetService("CoreGui") end
    end)
    if not sg.Parent then sg.Parent = LocalPlayer:WaitForChild("PlayerGui") end
end
local function plog(m) pcall(function() print("[MOL] "..m) end) end
local function SafeCall(fn, label)
    return function(...)
        local ok, err = pcall(fn, ...)
        if not ok then plog("Error ["..(label or "?").."]: "..tostring(err)) end
        return ok
    end
end

local function createLoadingScreen()
    local sg = Instance.new("ScreenGui")
    sg.Name = "MolLoad_" .. math.random(1000,9999)
    sg.ResetOnSpawn = false; sg.IgnoreGuiInset = true; sg.DisplayOrder = 999
    protectGui(sg)

    local bg = new("Frame", {Parent=sg, Size=UDim2.fromScale(1,1), BackgroundColor3=C.BG, BorderSizePixel=0})

    -- grid layers
    local function makeGrid(step, alpha, dir)
        local g = new("Frame", {Parent=bg, Size=UDim2.new(1,300,1,300),
            Position=UDim2.new(0,-150,0,-150), BackgroundTransparency=1, BorderSizePixel=0})
        local maxL = math.ceil(3000/step)
        for i = 0, maxL do
            new("Frame", {Parent=g, Size=UDim2.new(1,0,0,1), Position=UDim2.new(0,0,0,i*step),
                BackgroundColor3=C.DARK, BackgroundTransparency=1-alpha, BorderSizePixel=0})
            new("Frame", {Parent=g, Size=UDim2.new(0,1,1,0), Position=UDim2.new(0,i*step,0,0),
                BackgroundColor3=C.DARK, BackgroundTransparency=1-alpha, BorderSizePixel=0})
        end
        task.spawn(function()
            while g.Parent do
                for t = 0, 1, 0.05 do
                    if not g.Parent then return end
                    g.Position = UDim2.new(0, -150 + t*step*dir, 0, -150 + t*step*dir)
                    task.wait(0.03)
                end
            end
        end)
    end
    makeGrid(60, 0.06, 1)
    makeGrid(30, 0.03, -1)

    -- orbs
    local function makeOrb(pos, size, color, dur)
        local wrap = new("Frame", {Parent=bg, Size=UDim2.fromOffset(size, size),
            Position=pos, BackgroundTransparency=1, BorderSizePixel=0})
        for i = 8, 1, -1 do
            local layer = new("Frame", {Parent=wrap,
                Size=UDim2.fromOffset(size*(i/8), size*(i/8)),
                Position=UDim2.new(0.5,0,0.5,0), AnchorPoint=Vector2.new(0.5,0.5),
                BackgroundColor3=color, BackgroundTransparency=1-(0.15/i), BorderSizePixel=0})
            corner(layer, size)
        end
        task.spawn(function()
            local t0 = tick(); local bx, by = pos.X.Offset, pos.Y.Offset
            while wrap.Parent do
                local dt = tick() - t0
                wrap.Position = UDim2.new(pos.X.Scale, bx + math.sin(dt*(2*math.pi/dur)+size)*30,
                                          pos.Y.Scale, by + math.cos(dt*(2*math.pi/dur)+size)*25)
                task.wait(0.04)
            end
        end)
    end
    makeOrb(UDim2.new(0.08,0,0.10,0), 340, C.INDIGO, 18)
    makeOrb(UDim2.new(0.68,0,0.58,0), 300, C.PINK, 22)
    makeOrb(UDim2.new(0.30,0,0.78,0), 320, C.CYAN, 20)

    new("Frame", {Parent=bg, Size=UDim2.fromScale(1,1), BackgroundColor3=C.BG, BackgroundTransparency=0.85, BorderSizePixel=0})

    local center = new("Frame", {Parent=bg, Size=UDim2.new(0,420,0,340),
        Position=UDim2.new(0.5,0,0.5,0), AnchorPoint=Vector2.new(0.5,0.5), BackgroundTransparency=1})

    local logo = new("Frame", {Parent=center, Size=UDim2.fromOffset(80,80),
        Position=UDim2.new(0.5,0,0,0), AnchorPoint=Vector2.new(0.5,0), BackgroundColor3=C.DARK, BorderSizePixel=0})

        Text="MOL", TextColor3=C.WHITE, Font=Enum.Font.GothamBlack, TextSize=26})

        AnchorPoint=Vector2.new(0.5,0), BackgroundTransparency=1,
        Text="MOL HUB", TextColor3=C.TEXT, Font=Enum.Font.GothamBlack, TextSize=28})

    local loadStatusLbl = new("TextLabel", {Parent=center, Size=UDim2.new(1,0,0,20),
        Position=UDim2.new(0.5,0,0,132), AnchorPoint=Vector2.new(0.5,0),
        BackgroundTransparency=1, Text="Memuat...", TextColor3=C.TEXT2,
        Font=Enum.Font.GothamSemibold, TextSize=13})

        Position=UDim2.new(0.5,0,0,168), AnchorPoint=Vector2.new(0.5,0), BackgroundTransparency=1})

    for i = 1, 20 do
        local a = (i/20) * math.pi * 2
        local s = new("Frame", {Parent=spinner, Size=UDim2.fromOffset(5,5),
            Position=UDim2.new(0.5, math.cos(a)*16, 0.5, math.sin(a)*16),

        AnchorPoint=Vector2.new(0.5,0), BackgroundColor3=C.SHADOW, BorderSizePixel=0}).Parent = center

local function createToggle(parent, name, default, callback)
    local row = new("Frame", {Parent=parent, Size=UDim2.new(1,0,0,32),
        BackgroundTransparency=1, BorderSizePixel=0, ZIndex=2})
    local lbl = new("TextLabel", {Parent=row, Size=UDim2.new(1,-56,1,0), Position=UDim2.new(0,2,0,0),
        BackgroundTransparency=1, Text=name, TextColor3=C.TEXT,
        Font=Enum.Font.GothamBold, TextSize=12, TextXAlignment=Enum.TextXAlignment.Left, ZIndex=2})

    local swShadow = new("Frame", {Parent=row, Size=UDim2.fromOffset(44,22),
        Position=UDim2.new(1,-46,0.5,-11), BackgroundColor3=C.SHADOW, BorderSizePixel=0, ZIndex=2})
    corner(swShadow, 11)
    local sw = new("Frame", {Parent=row, Size=UDim2.fromOffset(44,22),
        Position=UDim2.new(1,-48,0.5,-13), BackgroundColor3=default and C.DARK or C.CARD,
        BorderSizePixel=0, ZIndex=3})
    corner(sw, 11); stroke(sw, C.BORDER, 2)

    local knob = new("Frame", {Parent=sw, Size=UDim2.fromOffset(14,14),
        Position=default and UDim2.new(1,-17,0.5,-7) or UDim2.new(0,3,0.5,-7),
        BackgroundColor3=C.WHITE, BorderSizePixel=0, ZIndex=4})
    corner(knob, 7); stroke(knob, C.BLACK, 1)

    local val = default
    local function setState(v)
        val = v
        tween(sw, 0.25, {BackgroundColor3 = v and C.DARK or C.CARD})
        tween(knob, 0.25, {Position = v and UDim2.new(1,-17,0.5,-7) or UDim2.new(0,3,0.5,-7)})
        if callback then pcall(callback, v) end
    end
    local btn = new("TextButton", {Parent=row, Size=UDim2.new(1,0,1,0),
        BackgroundTransparency=1, Text="", ZIndex=10})
    btn.MouseEnter:Connect(function() tween(sw, 0.15, {BackgroundColor3 = val and C.PURPLE or C.HOVER}) end)
    btn.MouseLeave:Connect(function() tween(sw, 0.15, {BackgroundColor3 = val and C.DARK or C.CARD}) end)
    btn.MouseButton1Click:Connect(function() setState(not val) end)

    -- Register for search
    table.insert(UI_ITEMS, {type="toggle", name=name, frame=row})
    return {Set=setState, Get=function() return val end}
end

        tween(btn, 0.15, {Position=UDim2.new(0,-1,0,-1)})

    btn.MouseLeave:Connect(function()
        tween(btn, 0.15, {Position=UDim2.new(0,0,0,0)})

    btn.MouseButton1Click:Connect(function()
        tween(btn, 0.08, {Position=UDim2.new(0,2,0,2)})

    end)
    table.insert(UI_ITEMS, {type="button", name=name, frame=wrap})

local function createSlider(parent, name, min, max, default, callback)
    local wrap = new("Frame", {Parent=parent, Size=UDim2.new(1,0,0,44),
        BackgroundTransparency=1, BorderSizePixel=0, ZIndex=2})
    new("TextLabel", {Parent=wrap, Size=UDim2.new(1,-50,0,16), Position=UDim2.new(0,2,0,0),
        BackgroundTransparency=1, Text=name, TextColor3=C.TEXT,
        Font=Enum.Font.GothamBold, TextSize=11, TextXAlignment=Enum.TextXAlignment.Left, ZIndex=2})
    local valLbl = new("TextLabel", {Parent=wrap, Size=UDim2.new(0,44,0,16), Position=UDim2.new(1,-46,0,0),
        BackgroundTransparency=1, Text=tostring(default), TextColor3=C.DARK,
        Font=Enum.Font.GothamBlack, TextSize=11, TextXAlignment=Enum.TextXAlignment.Right, ZIndex=2})

    local trackShadow = new("Frame", {Parent=wrap, Size=UDim2.new(1,0,0,10), Position=UDim2.new(0,3,0,26),
        BackgroundColor3=C.SHADOW, BorderSizePixel=0, ZIndex=1})
    corner(trackShadow, 5)
    local track = new("Frame", {Parent=wrap, Size=UDim2.new(1,0,0,10), Position=UDim2.new(0,1,0,24),
        BackgroundColor3=C.CARD, BorderSizePixel=0, ZIndex=2})
    corner(track, 5); stroke(track, C.BORDER, 2)
    local fill = new("Frame", {Parent=track, Size=UDim2.new((default-min)/(max-min),0,1,0),
        BackgroundColor3=C.DARK, BorderSizePixel=0, ZIndex=3})
    corner(fill, 4)
    local knob = new("Frame", {Parent=track, Size=UDim2.fromOffset(16,16),
        Position=UDim2.new((default-min)/(max-min),-8,0.5,-8),
        BackgroundColor3=C.WHITE, BorderSizePixel=0, ZIndex=5})
    corner(knob, 8); stroke(knob, C.BLACK, 1)

    local val = default
    local dragging = false
    local function updateFromX(mx)
        local rel = math.clamp((mx - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local v = math.floor(min + (max-min) * rel + 0.5)
        val = v; valLbl.Text = tostring(v)
        fill.Size = UDim2.new(rel, 0, 1, 0)
        knob.Position = UDim2.new(rel, -8, 0.5, -8)
        if callback then pcall(callback, v) end
    end
    local hit = new("TextButton", {Parent=wrap, Size=UDim2.new(1,0,1,0),
        BackgroundTransparency=1, Text="", ZIndex=20})
    hit.MouseButton1Down:Connect(function()
        dragging = true
        updateFromX(UserInputService:GetMouseLocation().X)
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging=false end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            updateFromX(input.Position.X)
        end
    end)
    table.insert(UI_ITEMS, {type="slider", name=name, frame=wrap})
    return {Set=function(v) updateFromX(track.AbsolutePosition.X + (v-min)/(max-min) * track.AbsoluteSize.X) end, Get=function() return val end}
end

    Color3.fromRGB(255,0,0), Color3.fromRGB(255,100,0), Color3.fromRGB(255,180,0), Color3.fromRGB(255,255,0),
    Color3.fromRGB(180,255,0), Color3.fromRGB(0,255,0), Color3.fromRGB(0,255,120), Color3.fromRGB(0,255,255),
    Color3.fromRGB(0,180,255), Color3.fromRGB(0,80,255), Color3.fromRGB(100,80,255), Color3.fromRGB(180,0,255),
    Color3.fromRGB(255,0,200), Color3.fromRGB(255,0,100), Color3.fromRGB(255,255,255), Color3.fromRGB(0,0,0),
}
local function createColorPicker(parent, name, default, callback)
    local wrap = new("Frame", {Parent=parent, Size=UDim2.new(1,0,0,32),
        BackgroundTransparency=1, BorderSizePixel=0, ZIndex=2})
    new("TextLabel", {Parent=wrap, Size=UDim2.new(1,-90,1,0), Position=UDim2.new(0,2,0,0),
        BackgroundTransparency=1, Text=name, TextColor3=C.TEXT,
        Font=Enum.Font.GothamBold, TextSize=11, TextXAlignment=Enum.TextXAlignment.Left, ZIndex=2})
    new("Frame", {Parent=wrap, Size=UDim2.fromOffset(50,24), Position=UDim2.new(1,-52,0.5,-10),
        BackgroundColor3=C.SHADOW, BorderSizePixel=0, ZIndex=1})
    local sel = new("TextButton", {Parent=wrap, Size=UDim2.fromOffset(50,24),
        Position=UDim2.new(1,-54,0.5,-12), BackgroundColor3=default,
        Text="", AutoButtonColor=false, BorderSizePixel=0, ZIndex=2})
    corner(sel, 7); stroke(sel, C.BORDER, 2)

    local open = false
    local colsPerRow = 8
    local rows = math.ceil(#PRESET_COLORS / colsPerRow)
    local gridW = colsPerRow * 26 + 10
    local gridH = rows * 26 + 10
    local gridShadow = new("Frame", {Parent=wrap, Size=UDim2.fromOffset(gridW, gridH),
        Position=UDim2.new(1,-gridW-2,0,32), BackgroundColor3=C.SHADOW,
        BorderSizePixel=0, Visible=false, ZIndex=49})
    corner(gridShadow, 7)
    local grid = new("Frame", {Parent=wrap, Size=UDim2.fromOffset(gridW, gridH),
        Position=UDim2.new(1,-gridW-4,0,30), BackgroundColor3=C.CARD,
        BorderSizePixel=0, Visible=false, ZIndex=50})
    corner(grid, 7); stroke(grid, C.BORDER, 2)
    for i, col in ipairs(PRESET_COLORS) do
        local r = math.floor((i-1)/colsPerRow)
        local ci = (i-1) % colsPerRow
        local sw = new("TextButton", {Parent=grid, Size=UDim2.fromOffset(22,22),
            Position=UDim2.new(0, 5 + ci*26, 0, 5 + r*26),
            BackgroundColor3=col, Text="", AutoButtonColor=false, BorderSizePixel=0, ZIndex=51})
        corner(sw, 5); stroke(sw, C.BLACK, 1)
        sw.MouseButton1Click:Connect(function()
            sel.BackgroundColor3 = col
            grid.Visible = false; gridShadow.Visible = false; open = false
            if callback then pcall(callback, col) end
        end)
    end
    sel.MouseButton1Click:Connect(function()
        open = not open
        grid.Visible = open; gridShadow.Visible = open
    end)
    table.insert(UI_ITEMS, {type="color", name=name, frame=wrap})
    return {Get=function() return sel.BackgroundColor3 end}
end

        b.MouseLeave:Connect(function() if activeTab ~= name then tween(b, 0.15, {BackgroundColor3=C.BG2}) end end)
        b.MouseButton1Click:Connect(function() showTab(name) end)

    end

            local delta = input.Position - dragStart

        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
    end)

            tween(winShadow, 0.25, {Size=UDim2.new(0,W,0,38)})
            sidebar.Visible = false; content.Visible = false; searchBar.Visible = false
        else
            tween(win, 0.25, {Size=UDim2.new(0,W,0,H)})
            tween(winShadow, 0.25, {Size=UDim2.new(0,W,0,H)})
            sidebar.Visible = true; content.Visible = true
        end
    end)

    -- SEARCH toggle
    local searchActive = false
    searchBtn.MouseButton1Click:Connect(function()
        searchActive = not searchActive
        SearchState.active = searchActive
        searchBar.Visible = searchActive
        if searchActive then
            tween(content, 0.2, {Position=UDim2.new(0,130,0,78)})
            tween(sidebar, 0.2, {Size=UDim2.new(0,130,1,-78)})
            content.Size = UDim2.new(1,-130,1,-78)
            searchBox:CaptureFocus()
        else
            tween(content, 0.2, {Position=UDim2.new(0,130,0,38)})
            sidebar.Size = UDim2.new(0,130,1,-38)
            content.Size = UDim2.new(1,-130,1,-38)
            searchBox.Text = ""
            SearchState.query = ""
            -- restore visibility
            for _, item in ipairs(UI_ITEMS) do
                if item.frame then item.frame.Visible = true end
            end
            for _, frame in ipairs(allTabFrames) do
                frame.Visible = (tabs[activeTab] and tabs[activeTab].container == frame) or false
            end
        end
    end)

    -- Search filtering
    searchBox:GetPropertyChangedSignal("Text"):Connect(function()
        local q = string.lower(searchBox.Text)
        SearchState.query = q
        if q == "" then
            for _, item in ipairs(UI_ITEMS) do
                if item.frame then item.frame.Visible = true end
            end
            return
        end
        for _, item in ipairs(UI_ITEMS) do
            if item.frame then
                local match = string.find(string.lower(item.name), q, 1, true) ~= nil
                item.frame.Visible = match
            end
        end
    end)

    searchClear.MouseButton1Click:Connect(function()
        searchBox.Text = ""
        SearchState.query = ""
        for _, item in ipairs(UI_ITEMS) do
            if item.frame then item.frame.Visible = true end
        end
    end)

    return {sg=sg, win=win, addTab=addTab, showTab=showTab, tabs=tabs}
end

local function IsEnemy(player)
    if player == LocalPlayer then return false end
    if not Config.TeamCheck then return true end
    if player.Team and LocalPlayer.Team then return player.Team ~= LocalPlayer.Team end
    return true
end
local function GetAimPart(player)
    if not player or not player.Character then return nil end
    local char = player.Character
    local part = char:FindFirstChild(Config.AimPart)
    if part and part:IsA("BasePart") then return part end
    for _, name in ipairs({"Head","UpperTorso","Torso","HumanoidRootPart","LowerTorso"}) do
        local p = char:FindFirstChild(name)
        if p and p:IsA("BasePart") then return p end
    end
    return nil
end
local function HasLineOfSight(part)
    if not part then return false end
    local origin = Camera.CFrame.Position
    local dir = part.Position - origin
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local ignore = {LocalPlayer.Character, Camera}
    if part.Parent then table.insert(ignore, part.Parent) end
    params.FilterDescendantsInstances = ignore
    params.IgnoreWater = true
    local result = workspace:Raycast(origin, dir, params)
    return result == nil or result.Instance == part or result.Instance:IsDescendantOf(part.Parent)
end
local function IsTargetValid(player)
    if not player or not player.Parent then return false end
    if not IsEnemy(player) then return false end
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    local part = GetAimPart(player)
    if not part then return false end
    if (part.Position - Camera.CFrame.Position).Magnitude > 5000 then return false end
    if Config.VisibleCheck and not HasLineOfSight(part) then return false end
    return true
end
local function AcquireTarget()
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local best, bestDist = nil, Config.FOV
    for _, player in ipairs(Players:GetPlayers()) do
        if IsEnemy(player) and player.Character then
            local part = GetAimPart(player)
            if part then
                local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen and sp.Z > 0 then
                    local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d < bestDist then
                        if not Config.VisibleCheck or HasLineOfSight(part) then
                            bestDist = d; best = player
                        end
                    end
                end
            end
        end
    end
    return best
end
local function GetPredictedPosition(part)
    if not Config.UsePrediction or not part then return part.Position end
    local char = part.Parent
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return part.Position end
    local vel = hrp.AssemblyLinearVelocity or hrp.Velocity or Vector3.zero
    local dist = (part.Position - Camera.CFrame.Position).Magnitude
    local travelTime = dist / 1000
    return part.Position + vel * travelTime * Config.Prediction
end
local function AimAt(player, instant)
    local part = GetAimPart(player)
    if not part then return end
    local targetPos = GetPredictedPosition(part)
    local goal = CFrame.new(Camera.CFrame.Position, targetPos)
    if instant or Config.Smoothness >= 1 then
        Camera.CFrame = goal
    else
        Camera.CFrame = Camera.CFrame:Lerp(goal, Config.Smoothness)
    end
end

local KILLER_WORDS = {"killer","monster","zombie","slasher","hunter","stalker","chaser","entity","beast","demon","ghost","phantom","murderer","assassin","vampire","werewolf"}
local function nameMatchesKiller(name)
    if not name then return false end
    local n = string.lower(tostring(name))
    for _, kw in ipairs(KILLER_WORDS) do if string.find(n, kw, 1, true) then return true end end
    return false
end
local KILLER_ATTRS = {"IsKiller","Killer","IsMonster","Monster","IsSlasher","Slasher","IsMurderer","Murderer","IsRole","Role","Team","RoleName","PlayerRole"}
local function checkAttributes(obj)
    if not obj then return false end
    for _, attr in ipairs(KILLER_ATTRS) do
        local ok, v = pcall(function() return obj:GetAttribute(attr) end)
        if ok and v ~= nil then
            if v == true then return true end
            if type(v) == "string" and nameMatchesKiller(v) then return true end
            if type(v) == "number" and v == 1 then return true end
        end
    end
    return false
end
local KILLER_TAGS = {"Killer","killer","Monster","monster","Slasher","slasher","Vampire","Murderer"}
local function checkTags(obj)
    if not obj then return false end
    for _, tag in ipairs(KILLER_TAGS) do
        local ok, has = pcall(function() return CollectionService:HasTag(obj, tag) end)
        if ok and has then return true end
    end
    return false
end
local function isKillerPlayer(player)
    if not player then return false end
    if checkAttributes(player) then return true end
    if checkTags(player) then return true end
    if nameMatchesKiller(player.Name) then return true end
    if nameMatchesKiller(player.DisplayName) then return true end
    local ls = player:FindFirstChild("leaderstats")
    if ls then
        for _, v in ipairs(ls:GetChildren()) do
            if nameMatchesKiller(v.Name) then return true end
            if v:IsA("StringValue") and nameMatchesKiller(v.Value) then return true end
        end
    end
    local char = player.Character
    if char then
        if checkAttributes(char) then return true end
        if checkTags(char) then return true end
    end
    return false
end

local function findRandomByKeywords(keywords)
    local found, seen = {}, {}
    local function matchName(n)
        n = string.lower(n or "")
        for _, kw in ipairs(keywords) do if string.find(n, kw, 1, true) then return true end end
        return false
    end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and matchName(obj.Name) and not seen[obj] then
            seen[obj] = true; table.insert(found, obj)
        end
        if obj:IsA("Model") and matchName(obj.Name) then
            local p = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
            if p and not seen[p] then seen[p] = true; table.insert(found, p) end
        end
    end
    if #found == 0 then return nil end
    return found[math.random(1, #found)]
end

task.spawn(function()
    RunService.RenderStepped:Connect(SafeCall(function()
        if Config.AimbotEnabled then
            if FOVCircle then
                FOVCircle.Visible = true
                FOVCircle.Position = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
                FOVCircle.Radius = Config.FOV
            end
        else
            if FOVCircle then FOVCircle.Visible = false end
        end
        local mouseHeld = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
        if not (Config.AimbotEnabled and (mouseHeld or Config.AutoAim)) then
            if currentTarget then ClearLock() end
            return
        end
        if currentTarget and not IsTargetValid(currentTarget) then ClearLock() end
        if not currentTarget or not Config.LockOn then
            local nt = AcquireTarget()
            if nt then currentTarget = nt end
        end
        if not currentTarget then return end
        local part = GetAimPart(currentTarget)
        if not part then ClearLock(); return end
        AimAt(currentTarget)
        if Config.AutoFire then
            local camLook = Camera.CFrame.LookVector
            local toTarget = (part.Position - Camera.CFrame.Position).Unit
            local dot = math.clamp(camLook:Dot(toTarget), -1, 1)
            local angle = math.deg(math.acos(dot))
            if angle <= 5 then
                local now = tick()
                if now - lastFire >= 0.1 then
                    lastFire = now
                    Camera.CFrame = CFrame.new(Camera.CFrame.Position, part.Position)
                    FireOnce()
                end
            end
        end
    end, "Aim"))
end)
