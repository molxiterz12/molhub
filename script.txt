local function protectGui(sg)
    pcall(function()
        if syn and syn.protect_gui then syn.protect_gui(sg); sg.Parent = game:GetService("CoreGui")
        elseif gethui then sg.Parent = gethui()
        else sg.Parent = game:GetService("CoreGui") end
    end)
    if not sg.Parent then sg.Parent = LocalPlayer:WaitForChild("PlayerGui") end
end
local function mk(class, props, children)
    local i = Instance.new(class)
    for k, v in pairs(props or {}) do if k ~= "Parent" then i[k] = v end end
    for _, c in ipairs(children or {}) do c.Parent = i end
    if props and props.Parent then i.Parent = props.Parent end
    return i
end
local function corner(i, r)
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, r or 8); c.Parent = i; return c
end
local function stroke(i, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color or Theme.Border; s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border; s.Parent = i; return s
end
local function tween(i, t, p)
    TweenService:Create(i, TweenInfo.new(t or 0.18, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), p):Play()
end

    local content = mk("Frame", {Parent=main, Size=UDim2.new(1,-130,1,-34), Position=UDim2.new(0,130,0,34), BackgroundColor3=Theme.Bg, BorderSizePixel=0, ClipsDescendants=true})

        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = main.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)

        sidebar.Visible = not minimized; content.Visible = not minimized
    end)

    local function showTab(name)
        activeTab = name
        for tn, tab in pairs(tabs) do
            tab.container.Visible = (tn == name)
            tween(tab.btn, 0.15, {BackgroundColor3 = tn == name and Theme.Element or Theme.Sidebar})
            tab.btnLbl.TextColor3 = tn == name and Theme.Text or Theme.SubText
        end
    end
    local function addTab(name)

        mk("UIListLayout", {Parent=container, Padding=UDim.new(0,6), SortOrder=Enum.SortOrder.LayoutOrder})
        mk("UIPadding", {Parent=container, PaddingTop=UDim.new(0,10), PaddingBottom=UDim.new(0,10), PaddingLeft=UDim.new(0,10), PaddingRight=UDim.new(0,10)})
        local b = mk("TextButton", {Parent=sidebar, Size=UDim2.new(1,0,0,32), BackgroundColor3=Theme.Sidebar, Text="", AutoButtonColor=false, BorderSizePixel=0})
        corner(b, 6)
        local lbl = mk("TextLabel", {Parent=b, Size=UDim2.new(1,-16,1,0), Position=UDim2.new(0,12,0,0), BackgroundTransparency=1, Text=name, TextColor3=Theme.SubText, Font=Enum.Font.GothamBold, TextSize=13, TextXAlignment=Enum.TextXAlignment.Left})
        tabs[name] = {container=container, btn=b, btnLbl=lbl}
        b.MouseEnter:Connect(function() if activeTab ~= name then tween(b, 0.12, {BackgroundColor3=Theme.Element}) end end)
        b.MouseLeave:Connect(function() if activeTab ~= name then tween(b, 0.12, {BackgroundColor3=Theme.Sidebar}) end end)
        b.MouseButton1Click:Connect(function() showTab(name) end)
        return container
    end
    return setmetatable({sg=sg, main=main, sidebar=sidebar, content=content, addTab=addTab, showTab=showTab, tabs=tabs}, Window)
end

local function CreateSlider(parent, name, min, max, default, callback, suffix)
    suffix = suffix or ""
    local wrap = mk("Frame", {Parent=parent, Size=UDim2.new(1,-4,0,44), BackgroundColor3=Theme.Card, BorderSizePixel=0})
    corner(wrap, 6); stroke(wrap, Theme.Border, 1)
    mk("TextLabel", {Parent=wrap, Size=UDim2.new(1,-70,0,20), Position=UDim2.new(0,10,0,5), BackgroundTransparency=1, Text=name, TextColor3=Theme.Text, Font=Enum.Font.GothamMedium, TextSize=12, TextXAlignment=Enum.TextXAlignment.Left})
    local valLbl = mk("TextLabel", {Parent=wrap, Size=UDim2.new(0,60,0,20), Position=UDim2.new(1,-66,0,5), BackgroundTransparency=1, Text=tostring(default)..suffix, TextColor3=Theme.Accent, Font=Enum.Font.GothamBold, TextSize=12, TextXAlignment=Enum.TextXAlignment.Right})
    local track = mk("Frame", {Parent=wrap, Size=UDim2.new(1,-20,0,6), Position=UDim2.new(0,10,0,30), BackgroundColor3=Theme.Element, BorderSizePixel=0})
    corner(track, 3)
    local fill = mk("Frame", {Parent=track, Size=UDim2.new((default-min)/(max-min),0,1,0), BackgroundColor3=Theme.Accent, BorderSizePixel=0})
    corner(fill, 3)
    local knob = mk("Frame", {Parent=track, Size=UDim2.fromOffset(12,12), Position=UDim2.new((default-min)/(max-min),-6,0.5,-6), BackgroundColor3=Color3.fromRGB(255,255,255), BorderSizePixel=0})
    corner(knob, 6)
    local dragging = false
    local function upd(mx)
        local rel = math.clamp((mx - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local v = math.floor(min + (max-min) * rel + 0.5)
        valLbl.Text = tostring(v)..suffix
        fill.Size = UDim2.new(rel, 0, 1, 0)
        knob.Position = UDim2.new(rel, -6, 0.5, -6)
        if callback then pcall(callback, v) end
    end
    local hit = mk("TextButton", {Parent=wrap, Size=UDim2.new(1,0,1,0), BackgroundTransparency=1, Text=""})
    hit.MouseButton1Down:Connect(function()
        dragging = true
        upd(UserInputService:GetMouseLocation().X)
    end)
    UserInputService.InputEnded:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging=false end end)
    UserInputService.InputChanged:Connect(function(input) if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then upd(input.Position.X) end end)
end

local function CreateColorPicker(parent, name, default, callback)
    local wrap = mk("Frame", {Parent=parent, Size=UDim2.new(1,-4,0,34), BackgroundColor3=Theme.Card, BorderSizePixel=0})
    corner(wrap, 6); stroke(wrap, Theme.Border, 1)
    mk("TextLabel", {Parent=wrap, Size=UDim2.new(1,-50,1,0), Position=UDim2.new(0,10,0,0), BackgroundTransparency=1, Text=name, TextColor3=Theme.Text, Font=Enum.Font.GothamMedium, TextSize=12, TextXAlignment=Enum.TextXAlignment.Left})

    corner(swatch, 5); stroke(swatch, Theme.Border, 1)
    local picker = mk("Frame", {Parent=wrap, Size=UDim2.new(0,180,0,96), Position=UDim2.new(1,-186,0,34), BackgroundColor3=Theme.Bg2, BorderSizePixel=0, Visible=false, ZIndex=100})
    corner(picker, 6); stroke(picker, Theme.Border, 1)
    local hueBar = mk("Frame", {Parent=picker, Size=UDim2.new(1,-16,0,14), Position=UDim2.new(0,8,0,8), BackgroundColor3=Color3.fromRGB(255,0,0), BorderSizePixel=0})
    corner(hueBar, 4)
    for i = 0, 40 do
        mk("Frame", {Parent=hueBar, Size=UDim2.new(1/40,0,1,0), Position=UDim2.new(i/40,0,0,0), BackgroundColor3=Color3.fromHSV(i/40,1,1), BorderSizePixel=0, ZIndex=101})
    end
    local hueKnob = mk("Frame", {Parent=hueBar, Size=UDim2.fromOffset(3,18), Position=UDim2.new(0,-1.5,0.5,-9), BackgroundColor3=Color3.fromRGB(255,255,255), BorderSizePixel=0, ZIndex=102})
    local svArea = mk("Frame", {Parent=picker, Size=UDim2.new(1,-16,0,44), Position=UDim2.new(0,8,0,28), BackgroundColor3=default, BorderSizePixel=0, ClipsDescendants=true})
    corner(svArea, 4)
    local svWhite = mk("Frame", {Parent=svArea, Size=UDim2.fromScale(1,1), BackgroundColor3=Color3.fromRGB(255,255,255), BorderSizePixel=0, ZIndex=101})
    local wg = Instance.new("UIGradient"); wg.Transparency = NumberSequence.new({NumberSequenceKeypoint.new(0,0),NumberSequenceKeypoint.new(1,1)}); wg.Parent = svWhite
    local svBlack = mk("Frame", {Parent=svArea, Size=UDim2.fromScale(1,1), BackgroundColor3=Color3.fromRGB(0,0,0), BorderSizePixel=0, ZIndex=102})
    local bg = Instance.new("UIGradient"); bg.Rotation = 90; bg.Transparency = NumberSequence.new({NumberSequenceKeypoint.new(0,1),NumberSequenceKeypoint.new(1,0)}); bg.Parent = svBlack
    local svCursor = mk("Frame", {Parent=svArea, Size=UDim2.fromOffset(10,10), Position=UDim2.new(1,-5,0,-5), BackgroundTransparency=1, BorderSizePixel=0, ZIndex=103})
    corner(svCursor, 5); stroke(svCursor, Color3.fromRGB(255,255,255), 2)
    local preview = mk("Frame", {Parent=picker, Size=UDim2.new(1,-16,0,14), Position=UDim2.new(0,8,0,78), BackgroundColor3=default, BorderSizePixel=0})
    corner(preview, 4); stroke(preview, Theme.Border, 1)
    local curH, curS, curV = 0.6, 1, 1
    local function updCol()
        local col = Color3.fromHSV(curH, curS, curV)
        preview.BackgroundColor3 = col; swatch.BackgroundColor3 = col
        svArea.BackgroundColor3 = Color3.fromHSV(curH, 1, 1)
        if callback then pcall(callback, col) end
    end
    local dH, dSV = false, false
    mk("TextButton", {Parent=hueBar, Size=UDim2.fromScale(1,1), BackgroundTransparency=1, Text="", ZIndex=105}).MouseButton1Down:Connect(function()
        dH = true
        local rel = math.clamp((UserInputService:GetMouseLocation().X - hueBar.AbsolutePosition.X) / hueBar.AbsoluteSize.X, 0, 1)

        local sx = math.clamp((UserInputService:GetMouseLocation().X - svArea.AbsolutePosition.X) / svArea.AbsoluteSize.X, 0, 1)
        local sy = math.clamp((UserInputService:GetMouseLocation().Y - svArea.AbsolutePosition.Y) / svArea.AbsoluteSize.Y, 0, 1)
        curS = sx; curV = 1 - sy; svCursor.Position = UDim2.new(sx,-5,sy,-5); updCol()
    end)
    UserInputService.InputEnded:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseButton1 then dH=false; dSV=false end end)
    UserInputService.InputChanged:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseMovement then return end
        if dH then
            local rel = math.clamp((input.Position.X - hueBar.AbsolutePosition.X) / hueBar.AbsoluteSize.X, 0, 1)
            curH = rel; hueKnob.Position = UDim2.new(rel,-1.5,0.5,-9); updCol()
        elseif dSV then
            local sx = math.clamp((input.Position.X - svArea.AbsolutePosition.X) / svArea.AbsoluteSize.X, 0, 1)
            local sy = math.clamp((input.Position.Y - svArea.AbsolutePosition.Y) / svArea.AbsoluteSize.Y, 0, 1)
            curS = sx; curV = 1 - sy; svCursor.Position = UDim2.new(sx,-5,sy,-5); updCol()
        end
    end)
    swatch.MouseButton1Click:Connect(function() picker.Visible = not picker.Visible end)
end

local currentTarget = nil

    if not p or not p.Character then return nil end
    return p.Character:FindFirstChild("HumanoidRootPart")

local function IsEnemy(player)
    if player == LocalPlayer then return false end
    if not Config.TeamCheck then return true end
    if player.Team and LocalPlayer.Team then return player.Team ~= LocalPlayer.Team end
    return true
end
local function GetAimPart(player)
    if not player or not player.Character then return nil end
    local char = player.Character
    local part = char:FindFirstChild(Config.AimPart)
    if part and part:IsA("BasePart") then return part end
    for _, name in ipairs({"Head","UpperTorso","Torso","HumanoidRootPart","LowerTorso"}) do
        local p = char:FindFirstChild(name)
        if p and p:IsA("BasePart") then return p end
    end
    return nil
end
local function HasLineOfSight(part)
    if not part then return false end
    local origin = Camera.CFrame.Position
    local dir = part.Position - origin
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local ignore = {LocalPlayer.Character, Camera}
    if part.Parent then table.insert(ignore, part.Parent) end
    params.FilterDescendantsInstances = ignore
    params.IgnoreWater = true
    local result = workspace:Raycast(origin, dir, params)
    return result == nil or result.Instance == part or result.Instance:IsDescendantOf(part.Parent)
end
local function IsTargetValid(player)
    if not player or not player.Parent then return false end
    if not IsEnemy(player) then return false end
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    local part = GetAimPart(player)
    if not part then return false end
    if (part.Position - Camera.CFrame.Position).Magnitude > 5000 then return false end
    if Config.VisibleCheck and not HasLineOfSight(part) then return false end
    return true
end
local function AcquireTarget()
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local best, bestDist = nil, Config.FOV
    for _, player in ipairs(Players:GetPlayers()) do
        if IsEnemy(player) and player.Character then
            local part = GetAimPart(player)
            if part then
                local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen and sp.Z > 0 then
                    local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d < bestDist then
                        if not Config.VisibleCheck or HasLineOfSight(part) then
                            bestDist = d; best = player
                        end
                    end
                end
            end
        end
    end
    return best
end
local function GetPredictedPosition(part)
    if not Config.UsePrediction or not part then return part.Position end
    local char = part.Parent
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return part.Position end
    local vel = hrp.AssemblyLinearVelocity or hrp.Velocity or Vector3.zero
    local dist = (part.Position - Camera.CFrame.Position).Magnitude
    local travelTime = dist / 1000
    return part.Position + vel * travelTime * Config.Prediction
end
local function AimAt(player, instant)
    local part = GetAimPart(player)
    if not part then return end
    local targetPos = GetPredictedPosition(part)
    local goal = CFrame.new(Camera.CFrame.Position, targetPos)
    if instant or Config.Smoothness >= 1 then
        Camera.CFrame = goal
    else
        Camera.CFrame = Camera.CFrame:Lerp(goal, Config.Smoothness)
    end
end
local function FireOnce()
    pcall(function()
        if mouse1click then mouse1click()
        elseif mouse1press and mouse1release then
            mouse1press(); task.wait(0.02); mouse1release()
        end
    end)
end

local KILLER_WORDS = {"killer","monster","zombie","slasher","hunter","stalker","chaser","entity","beast","demon","ghost","phantom","murderer","assassin","vampire","werewolf"}
local function nameMatchesKiller(name)
    if not name then return false end
    local n = string.lower(tostring(name))
    for _, kw in ipairs(KILLER_WORDS) do if string.find(n, kw, 1, true) then return true end end
    return false
end
local KILLER_ATTRS = {"IsKiller","Killer","IsMonster","Monster","IsSlasher","Slasher","IsMurderer","Murderer","IsRole","Role","Team","RoleName","PlayerRole"}
local function checkAttributes(obj)
    if not obj then return false end
    for _, attr in ipairs(KILLER_ATTRS) do
        local ok, v = pcall(function() return obj:GetAttribute(attr) end)
        if ok and v ~= nil then
            if v == true then return true end
            if type(v) == "string" and nameMatchesKiller(v) then return true end
            if type(v) == "number" and v == 1 then return true end
        end
    end
    return false
end
local KILLER_TAGS = {"Killer","killer","Monster","monster","Slasher","slasher","Vampire","Murderer"}
local function checkTags(obj)
    if not obj then return false end
    for _, tag in ipairs(KILLER_TAGS) do
        local ok, has = pcall(function() return CollectionService:HasTag(obj, tag) end)
        if ok and has then return true end
    end
    return false
end
local function isKillerPlayer(player)
    if not player then return false end
    if checkAttributes(player) then return true end
    if checkTags(player) then return true end
    if nameMatchesKiller(player.Name) then return true end
    if nameMatchesKiller(player.DisplayName) then return true end
    local ls = player:FindFirstChild("leaderstats")
    if ls then
        for _, v in ipairs(ls:GetChildren()) do
            if nameMatchesKiller(v.Name) then return true end
            if v:IsA("StringValue") and nameMatchesKiller(v.Value) then return true end
        end
    end
    return false
end

        end
    end

local function forceTP(targetPart)

        tpTo(targetPart); task.wait(0.05)
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp and (hrp.Position - targetPart.Position).Magnitude < 10 then return true end
    end

local function findRandomByKeywords(keywords)
    local found, seen = {}, {}
    local function addPart(p)
        if p and p:IsA("BasePart") and not seen[p] then seen[p]=true; table.insert(found,p) end
    end
    local function matchName(n)
        n = string.lower(n or "")
        for _, kw in ipairs(keywords) do if string.find(n, kw, 1, true) then return true end end
        return false
    end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and matchName(obj.Name) then addPart(obj) end
        if obj:IsA("Model") and matchName(obj.Name) then
            local p = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
            if p then addPart(p) end
        end
    end
    if #found == 0 then return nil end
    return found[math.random(1, #found)]
end

    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end
    flySavedWS = hum.WalkSpeed; flySavedJP = hum.JumpPower; flySavedAR = hum.AutoRotate
    hum.PlatformStand = true; hum.AutoRotate = false; hum.WalkSpeed = 0; hum.JumpPower = 0
    for _, m in ipairs(hrp:GetChildren()) do if m.Name=="MolFlyBV" or m.Name=="MolFlyBG" then m:Destroy() end end
    flyBV = Instance.new("BodyVelocity")
    flyBV.Name="MolFlyBV"; flyBV.MaxForce=Vector3.new(math.huge,math.huge,math.huge)
    flyBV.P=1250; flyBV.Velocity=Vector3.zero; flyBV.Parent=hrp
    flyBG = Instance.new("BodyGyro")
    flyBG.Name="MolFlyBG"; flyBG.MaxTorque=Vector3.new(math.huge,math.huge,math.huge)
    flyBG.P=100000; flyBG.D=1000; flyBG.CFrame=hrp.CFrame; flyBG.Parent=hrp
    if flyConn then flyConn:Disconnect() end
    flyConn = RunService.RenderStepped:Connect(SafeCall(function()
        if not states.Fly then return end
        if not hrp.Parent or not flyBV.Parent or not flyBG.Parent then return end
        hum.PlatformStand=true; hum.WalkSpeed=0; hum.JumpPower=0; hum.AutoRotate=false
        local camCF = Camera.CFrame
        local look = camCF.LookVector
        local right = Vector3.new(camCF.RightVector.X, 0, camCF.RightVector.Z).Unit
        local move = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move += look end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move -= look end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move -= right end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move += right end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move -= Vector3.new(0,1,0) end
        if move.Magnitude > 0.01 then
            flyBV.Velocity = move.Unit * Config.FlySpeed
        else
            flyBV.Velocity = Vector3.zero
        end
        flyBG.CFrame = camCF
    end, "FlyLoop"))
end
local function stopFly()
    if flyConn then flyConn:Disconnect(); flyConn=nil end
    if flyBV then flyBV:Destroy(); flyBV=nil end
    if flyBG then flyBG:Destroy(); flyBG=nil end
    local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.PlatformStand=false
        hum.WalkSpeed = flySavedWS or 16
        hum.JumpPower = flySavedJP or 50
        hum.AutoRotate = flySavedAR ~= false
    end
end
